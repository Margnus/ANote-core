import CryptoKit
import Foundation
import NoteCanvasPrototype

func measured<T>(_ label: String, _ block: () throws -> T) rethrows -> T {
    let start = ContinuousClock.now
    let value = try block()
    let interval = start.duration(to: .now).components
    let ms = Double(interval.seconds) * 1000 + Double(interval.attoseconds) / 1e15
    print("\(label)=\(ms)")
    return value
}

let mode = CommandLine.arguments[1]
let input = URL(fileURLWithPath: CommandLine.arguments[2])
let content = try String(contentsOf: input, encoding: .utf8)
let checkpoint = measured("\(mode)_read") {
    PreviewRecoveryCodec.shared.read(text: content).snapshot!
}
for i in 0..<3 {
    let text = try measured("\(mode)_encode_\(i)") { try checkpoint.encode() }
    let hash = SHA256.hash(data: Data(text.utf8)).map { String(format: "%02x", $0) }.joined()
    print("\(mode)_bytes=\(text.utf8.count) sha256=\(hash)")
}
