import subprocess,os,json,hashlib,time

def save(p,v):

def fetch(i):
 start=i*SIZE//6;end=(i+1)*SIZE//6;position=start;attempts=[]
 for attempt in range(3):
  if position==end:break
  began=position;headers=W/('range-'+str(i)+'-'+str(attempt)+'.headers');err=W/('range-'+str(i)+'-'+str(attempt)+'.stderr.log')
  with err.open('wb') as error:
   child=subprocess.Popen(['curl','--fail','--location','--range',str(position)+'-'+str(end-1),'--max-filesize',str(end-position),'--max-time','1800','--retry','0','--silent','--show-error','--dump-header',str(headers),URL],stdout=subprocess.PIPE,stderr=error)
   checkpoint=position+8*1024*1024

   code=child.wait();assert code in (0,28),(i,code,err.read_text())
  assert 'content-range: bytes '+str(began)+'-'+str(end-1)+'/'+str(SIZE) in headers.read_text().lower()
  attempts.append({'requested':[began,end-1],'received':position-began,'returncode':code})
  save(W/('range-'+str(i)+'.json'),{'range':[start,end-1],'received_bytes':position-start,'attempts':attempts})
