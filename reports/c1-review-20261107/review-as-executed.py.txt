def git(*args, cwd=ROOT):
    return subprocess.check_output(['git', *args], cwd=cwd)

def sha(data):
    return hashlib.sha256(data).hexdigest()

def load(path):
    return json.loads(path.read_text(encoding='utf-8'))

def save(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')

def report(name):
    return load(Path('reports') / (name + '.json'))

names = sorted({n for reports in groups.values() for n in reports} | {'c1-evidence-audit'})
reviewed = {}
for name in names:
    path = 'reports/' + name + '.json'
    authored = git('log','-1','--format=%H','--',path).decode().strip()
    data = Path(path).read_bytes()
    assert git('show',authored+':'+path)==data, path
    info = json.loads(data)
    binding = {}
    for key in ('source_sha256','source_and_artifact_sha256','source_and_evidence_sha256',
                'source_and_retained_evidence_sha256','evidence_sha256','parent_receipt_sha256'):
        binding.update(info.get(key, {}))
    counts = dict(matched=0, changed=0, missing=0)
    examples = []
    for source, expected in sorted(binding.items()):
        candidate=Path(source)
        state = ('missing' if not candidate.is_file() else
                 'matched' if sha(candidate.read_bytes())==expected else 'changed')
        counts[state] += 1
        if state!='matched' and len(examples)<5:
            examples.append({'path':source,'state':state})
    reviewed[name] = {'report':path,'commit':authored,'sha256':sha(data),
        'original_status':info.get('status'),'current_path_binding_counts':counts,
        'first_differences':examples}

# Match the recent product-consumer inputs independently of test/runner changes.
def exact_product(name, prefixes):
    sources = {p:h for p,h in report(name)['source_sha256'].items() if p.startswith(prefixes)}
    assert sources and all(sha(Path(p).read_bytes()) == h for p,h in sources.items()), name
    return len(sources)
