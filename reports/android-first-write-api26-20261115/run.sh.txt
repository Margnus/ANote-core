#!/bin/bash
set -euo pipefail
run=$(mktemp -d "$PWD/build/api26-atomic-boot.XXXXXX")
echo "Evidence: $run"
sdk="$HOME/Library/Android/sdk"
export JAVA_HOME="$HOME/Library/Java/JavaVirtualMachines/jbr-17.0.14/Contents/Home"
export ANDROID_HOME="$sdk" ANDROID_AVD_HOME="$run/avd" ANDROID_EMULATOR_HOME="$run" ANDROID_USER_HOME="$run"
mkdir -p "$ANDROID_AVD_HOME"
adb="$sdk/platform-tools/adb"
name=NoteCoreAtomicCut26
serial=emulator-5582
avd="$ANDROID_AVD_HOME/$name.avd"
mkdir -p "$avd"
printf 'avd.ini.encoding=UTF-8\npath=%s\ntarget=android-26\n' "$avd" > "$ANDROID_AVD_HOME/$name.ini"
sed -e "s/AvdId = .*/AvdId = $name/" -e "s/avd.ini.displayname = .*/avd.ini.displayname = $name/" -e "s@^image.sysdir.1 = .*@image.sysdir.1 = $PWD/build/api26-probe/image/arm64-v8a/@" -e 's/^tag.id = .*/tag.id = default/' -e 's/^tag.display = .*/tag.display = Default Android/' -e 's/^disk.dataPartition.size = .*/disk.dataPartition.size = 2147483648/' .android-avd/NoteCorePreview.avd/config.ini > "$avd/config.ini"
"$sdk/emulator/emulator" -avd "$name" -port 5582 -no-window -no-audio -no-snapshot -gpu swiftshader_indirect > "$run/emulator.log" 2>&1 &
pid=$!
trap '"$adb" -s "$serial" emu kill >/dev/null 2>&1 || kill "$pid" 2>/dev/null || true; wait "$pid" 2>/dev/null || true; rm -rf "$ANDROID_AVD_HOME"' EXIT
for i in $(seq 1 100); do
  if ! kill -0 "$pid" 2>/dev/null; then tail -60 "$run/emulator.log"; exit 1; fi
  if test "$("$adb" -s "$serial" shell getprop sys.boot_completed 2>/dev/null || true)" = 1; then break; fi
  sleep 2
done
test "$("$adb" -s "$serial" shell getprop sys.boot_completed | tr -d '\r')" = 1
test "$("$adb" -s "$serial" shell getprop ro.kernel.qemu.avd_name | tr -d '\r')" = "$name"
"$adb" -s "$serial" shell getprop ro.product.cpu.abi > "$run/abi.txt"
"$adb" -s "$serial" shell getenforce > "$run/selinux.txt"
"$adb" -s "$serial" shell getprop ro.build.fingerprint > "$run/fingerprint.txt"
python3 tools/verify_android_atomic_cut.py --adb "$adb" --serial "$serial" --isolated-api26
PYTHONDONTWRITEBYTECODE=1 python3 build/api26-probe/process.py "$run" "$adb"
fixture=$(find reports/view-pending-process-20261106 -name prepare.ncp -exec base64 -i {} \;)
failed=0
for case_name in "$@"; do
 extra=()
 if test "$case_name" = recovery_open_budget; then extra=(-e recovery_open_base64 "$fixture"); fi
 "$adb" -s "$serial" shell am instrument -w -r -e notecore_case "$case_name" ${extra[@]+"${extra[@]}"} dev.notecore.preview.test/dev.notecore.app.ViewportInstrumentation > "$run/$case_name.log"
 "$adb" -s "$serial" logcat -d -b crash > "$run/$case_name-crash.log"
 if ! rg -q 'INSTRUMENTATION_CODE: -1' "$run/$case_name.log"; then cat "$run/$case_name.log" "$run/$case_name-crash.log"; failed=1; continue; fi
 echo "$case_name passed"
done
exit "$failed"
